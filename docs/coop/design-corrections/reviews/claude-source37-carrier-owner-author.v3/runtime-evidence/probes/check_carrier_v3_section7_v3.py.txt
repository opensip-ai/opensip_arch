# SQLite types values dynamically: TEXT affinity keeps a BLOB, INTEGER affinity keeps a non-integral REAL, and
# length() and GLOB stop at an embedded NUL. Every column's storage class is enforced by an explicit typeof() guard
# or by exact enumeration, and every hex-bearing column is an exact whole TEXT value (typeof text, character length,
# instr(column, char(0)) = 0, prefix, whole-hex guard). Every column is exercised under the UTF-8, UTF-16le and
# UTF-16be database encodings: canonical lawful values must be admitted in their exact storage class and hostile
# values must never be stored outside the column law. Read-back uses typeof() and CAST(... AS BLOB) decoded in the
# observed database encoding, so no admitted value can crash the observation.

def _s37_run(script, steps):
    c = sqlite3.connect(':memory:', isolation_level=None)
    c.executescript(script)
    out = []
    for sql, args in steps:
        try:
            c.execute(sql, args)
            out.append(True)
        except sqlite3.Error:
            out.append(False)
    c.close()
    return out


def _s37_try(c, sql, args):
    try:
        c.execute(sql, args)
        return True
    except sqlite3.Error:
        return False

def _s37_hexlaw(prefix, n, nullable):
    pat = re.compile(re.escape(prefix).encode() + b'[0-9a-f]{%d}' % n)
    return lambda t, b: (t == 'null' and nullable) or (t == 'text' and pat.fullmatch(b) is not None)


def _s37_intlaw(pred, nullable=False):
    return lambda t, b: (t == 'null' and nullable) or (t == 'integer' and pred(int(b)))


def _s37_enumlaw(values, nullable=False):
    enc = {v.encode() for v in values}
    return lambda t, b: (t == 'null' and nullable) or (t == 'text' and b in enc)


def _s37_textlaw(nullable):
    return lambda t, b: (t == 'null' and nullable) or t == 'text'

def _s37_hex_hostile(prefix, v):
    n, mid, p = len(v), len(v) // 2, len(prefix)
    rows = [('uppercase-last', '?', v[:-1] + 'A'), ('nonhex-last', '?', v[:-1] + 'g'),
            ('nonhex-tail', '?', v[:p + 1] + 'Z' * (n - p - 1)), ('nul-suffix-Z', '?', v + '\x00Z'),
            ('nul-suffix', '?', v + '\x00'), ('nul-middle', '?', v[:mid] + '\x00' + v[mid + 1:]),
            ('ascii-hex-blob', '?', v.encode()), ('blob-nul-suffix', '?', v.encode() + b'\x00Z'),
            ('nonhex-blob', '?', b'Z' * n), ('invalid-utf8-text', 'CAST(? AS TEXT)', v[:-1].encode() + b'\xc0'),
            ('invalid-utf8-text-nul', 'CAST(? AS TEXT)', v.encode() + b'\x00\xff'),
            ('multibyte', '?', v[:-1] + 'é'), ('trailing-newline', '?', v + '\n'), ('short', '?', v[:-1]),
            ('long', '?', v + 'a'), ('integer', '?', 12345), ('real', '?', 1.5)]
    if prefix:
        rows += [('uppercase-prefix', '?', prefix.upper() + v[p:]), ('wrong-prefix', '?', 'x' * p + v[p:]),
                 ('nul-after-prefix', '?', prefix + '\x00' + v[p + 1:])]
    return rows


def _s37_hex_extra(v, enc):
    """Encoding-specific malformed TEXT: overlong and surrogate UTF-8; lone surrogate, NUL unit and odd byte in UTF-16."""
    head = v[:-1]
    if enc == 'UTF-8':
        raw = [('overlong-nul', head.encode() + b'\xc0\x80'), ('overlong-hex-digit', head.encode() + b'\xc0\xb0'),
               ('cesu-surrogate', head.encode() + b'\xed\xa0\x80'), ('lone-continuation', head.encode() + b'\x80')]
    else:
        codec = _S37_CODEC[enc]
        raw = [('lone-surrogate', head.encode(codec) + (b'\x00\xd8' if enc == 'UTF-16le' else b'\xd8\x00')),
               ('nul-code-unit', v.encode(codec) + '\x00'.encode(codec)), ('odd-trailing-byte', v.encode(codec) + b'\x41')]
    return [(l_, 'CAST(? AS TEXT)', b_) for l_, b_ in raw] + [('fullwidth-digit', '?', head + '０'),
                                                            ('arabic-indic-digit', '?', head + '٠')]


def _s37_int_hostile(v):
    return [('text-integral', '?', str(v)), ('real-integral', '?', float(v)), ('real-fraction', '?', v + 0.5),
            ('text-fraction', '?', str(v) + '.5'), ('blob', '?', bytes([v])), ('text-nul', '?', str(v) + '\x00')]


def _s37_enum_hostile(v):
    return [('blob', '?', v.encode()), ('nul-suffix', '?', v + '\x00'), ('nul-suffix-Z', '?', v + '\x00Z'),
            ('lowercase', '?', v.lower() if v.lower() != v else v.upper()), ('integer', '?', 7)]


def _s37_text_hostile(v):
    return [('blob', '?', v.encode()), ('integer', '?', 7), ('real', '?', 2.5), ('with-nul', '?', v + '\x00x')]

def _s37_insert(table, base, column, expr, value, encoding='UTF-8'):
    """Insert one row in a database of `encoding`; return ((typeof, utf8 value bytes) or None if refused, observed encoding)."""
    c = sqlite3.connect(':memory:', isolation_level=None)
    try:
        c.execute("PRAGMA encoding = '%s'" % encoding)
        c.executescript(_S37_AC if table == 'attempt_custody' else ddl3)
        if table == 'grant_journal_v3':
            c.execute(_S37_CF, _S37_FRESH)
        observed = c.execute('PRAGMA encoding').fetchone()[0]
        cols = list(_S37_BASE[base])
        c.execute('INSERT INTO %s (%s) VALUES (%s)' % (table, ','.join(cols), ','.join(expr if k == column else '?' for k in cols)),
                  [value if k == column else _S37_BASE[base][k] for k in cols])
        t, b = c.execute('SELECT typeof(%s), CAST(%s AS BLOB) FROM %s' % (column, column, table)).fetchone()
        if t in ('null', 'blob'):
            return (t, b), observed
        try:
            return (t, b.decode(_S37_CODEC[observed]).encode('utf-8')), observed
        except UnicodeDecodeError:
            return (t, b'\xff<undecodable>'), observed
    except sqlite3.Error:
        return None, None
    finally:
        c.close()
